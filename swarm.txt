# 1. Launch the 3 VMs
multipass launch --name node1 --cpus 1 --memory 1G --disk 5G
multipass launch --name node2 --cpus 1 --memory 1G --disk 5G
multipass launch --name node3 --cpus 1 --memory 1G --disk 5G

# 2. Install Docker on all 3 nodes
"node1", "node2", "node3" | ForEach-Object {
    multipass exec $_ -- bash -c "curl -fsSL https://get.docker.com | sh"
    multipass exec $_ -- sudo usermod -aG docker ubuntu
}

# 3. Get Manager IP
$MANAGER_IP = (multipass info node1 | Select-String "IPv4").Line.Split(":")[1].Trim()
Write-Host "Manager IP: $MANAGER_IP"

# 4. Initialize Swarm on node1
multipass exec node1 -- sudo docker swarm init --advertise-addr $MANAGER_IP

# 5. Extract Join Command
$JOIN_OUTPUT = multipass exec node1 -- sudo docker swarm join-token worker
$JOIN_CMD = ($JOIN_OUTPUT | Select-String "docker swarm join").Line.Trim()

# 6. Join Workers
multipass exec node2 -- sudo $JOIN_CMD
multipass exec node3 -- sudo $JOIN_CMD

# 7. Check Cluster Status
multipass exec node1 -- sudo docker node ls
